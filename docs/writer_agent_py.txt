from agentscope.agent import AgentBase
from agentscope.message import Msg
from agentscope.model import OpenAIChatModel

class WriterAgent(AgentBase):
    """
    Specialized agent for generating character backstories and personalities.
    """
    def __init__(self, name: str, model_config: dict):
        super().__init__()
        self.name = name
        self.sys_prompt = """
You are the Lead Writer for 'Scarab of Ra'.
Your job is to create deep, compelling characters for a Lovecraftian archaeological expedition (1920s Egypt).
Focus on:
- Psychological flaws.
- Hidden agendas.
- Connections to the occult.
- Period-accurate details.
"""
        self.model = OpenAIChatModel(
            model_name=model_config.get("model_name", "local-model"),
            api_key=model_config.get("api_key", "lm-studio"),
            client_kwargs={"base_url": model_config.get("api_base", "http://localhost:1234/v1")},
            generate_kwargs=model_config.get("generate_args", {})
        )

    async def reply(self, x: Msg = None) -> Msg:
        # Input: JSON-like prompt with constraints (Name, Class, Tone)
        prompt = f"""
{self.sys_prompt}

Task: Generate character details.
Constraints:
{x.content}

Output format (JSON):
{{
  "backstory": "...",
  "personality": ["trait1", "trait2"],
  "appearance": "...",
  "lineage": "..."
}}
"""
        messages = [{"role": "user", "content": prompt}]
        try:
            res = self.model(messages)
            content = res.text if hasattr(res, "text") else str(res)
            # Cleanup JSON markdown
            if "```json" in content:
                content = content.split("```json")[1].split("```")[0].strip()
            elif "```" in content:
                content = content.split("```")[0].strip()
        except Exception as e:
            content = "{ \"error\": \"" + str(e) + "\" }"
            
        return Msg(name=self.name, content=content, role="assistant")
