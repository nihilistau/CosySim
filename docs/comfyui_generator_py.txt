import json
import urllib.request
import urllib.parse
import time
import os
import random
import websocket # requires `pip install websocket-client`
import uuid

class ImageGenerator:
    """
    Standalone Python wrapper for ComfyUI workflow.
    """
    def __init__(self, workflow_path: str = None, server: str = "127.0.0.1:8188"):
        self.server_address = server
        self.client_id = str(uuid.uuid4())
        self.ws = websocket.WebSocket()
        self.ws.connect("ws://{}/ws?clientId={}".format(self.server_address, self.client_id))
        
        if workflow_path and os.path.exists(workflow_path):
            with open(workflow_path, "r", encoding="utf-8") as f:
                self.workflow = json.load(f)
        else:
            self.workflow = None # Will load default or fail if not provided

    def generate(self, positive_prompt: str, negative_prompt: str = "", resolution: str = "1024x1024", steps: int = 20, cfg: float = 7.0):
        if not self.workflow:
            raise ValueError("No workflow loaded.")
            
        workflow = self.workflow.copy()
        
        # Inject parameters into standard nodes
        # Note: Node IDs must match your specific workflow.
        # This implementation assumes standard ComfyUI default node IDs or the user's specific workflow.
        # User mentioned nodes: 86 (Pos), 4 (Neg), 5 (Empty Latent?), 71 (Resolution?), 1 (Checkpoint)
        
        # We need to map these dynamically or use a robust search, but for now we follow the user's hint
        # "Node 86: Main positive prompt", "Node 4: Main negative prompt"
        
        try:
            # Positive Prompt
            if "86" in workflow:
                workflow["86"]["inputs"]["text"] = positive_prompt
            elif "6" in workflow: # Fallback to standard default
                workflow["6"]["inputs"]["text"] = positive_prompt
                
            # Negative Prompt
            if "4" in workflow and "inputs" in workflow["4"] and "text" in workflow["4"]["inputs"]: # User said 4 is neg?
                # Usually 4 is checkpoint in default, 7 is negative. 
                # User's guide says: "4 Main negative prompt". We trust the user's workflow.
                workflow["4"]["inputs"]["text"] = negative_prompt
            elif "7" in workflow: # Fallback
                workflow["7"]["inputs"]["text"] = negative_prompt
                
            # Seed (Randomize)
            if "5" in workflow: # KSampler or similar
                if "seed" in workflow["5"]["inputs"]:
                    workflow["5"]["inputs"]["seed"] = random.randint(1, 1000000000000)
            if "3" in workflow: # KSampler default
                if "seed" in workflow["3"]["inputs"]:
                    workflow["3"]["inputs"]["seed"] = random.randint(1, 1000000000000)

        except KeyError as e:
            print(f"Warning: Could not inject prompt into workflow. Key error: {e}")

        prompt_id = self.queue_prompt(workflow)["prompt_id"]
        return self.track_progress(prompt_id)

    def queue_prompt(self, prompt_workflow):
        p = {"prompt": prompt_workflow, "client_id": self.client_id}
        data = json.dumps(p).encode('utf-8')
        req = urllib.request.Request("http://{}/prompt".format(self.server_address), data=data)
        return json.loads(urllib.request.urlopen(req).read())

    def track_progress(self, prompt_id):
        output_images = {}
        while True:
            out = self.ws.recv()
            if isinstance(out, str):
                message = json.loads(out)
                if message['type'] == 'executing':
                    data = message['data']
                    if data['node'] is None and data['prompt_id'] == prompt_id:
                        break # Execution finished
            else:
                continue # Binary data
                
        # Fetch history
        with urllib.request.urlopen("http://{}/history/{}".format(self.server_address, prompt_id)) as response:
            history = json.loads(response.read())
            
        outputs = history[prompt_id]['outputs']
        result_files = []
        
        for node_id in outputs:
            node_output = outputs[node_id]
            if 'images' in node_output:
                for image in node_output['images']:
                    image_data = self.get_image(image['filename'], image['subfolder'], image['type'])
                    result_files.append({
                        "filename": image['filename'],
                        "data": image_data
                    })
        return result_files

    def get_image(self, filename, subfolder, folder_type):
        data = {"filename": filename, "subfolder": subfolder, "type": folder_type}
        url_values = urllib.parse.urlencode(data)
        with urllib.request.urlopen("http://{}/view?{}".format(self.server_address, url_values)) as response:
            return response.read()

    def save_images(self, result_files, output_dir):
        if not os.path.exists(output_dir):
            os.makedirs(output_dir)
        
        saved_paths = []
        for img in result_files:
            path = os.path.join(output_dir, img['filename'])
            with open(path, "wb") as f:
                f.write(img['data'])
            saved_paths.append(path)
        return saved_paths
