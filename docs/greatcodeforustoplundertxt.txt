The Reactive Streaming Pipeline
This script assumes your Agent is writing to ./streams/session_1.parquet. The monitor watches the file, reads new chunks, evaluates quality, and pipes "good" audio to your speakers.
python
import time
import subprocess
import os
import pyarrow.parquet as pq
from watchdog.observers import Observer
from watchdog.events import FileSystemEventHandler

# 1. SETUP THE FFPLAY PIPE (The Voice)
# Adjust -ar (rate) and -ac (channels) to match your TTS output
FFPLAY_CMD = ['ffplay', '-i', 'pipe:0', '-f', 's16le', '-ar', '24000', '-ac', '1', '-nodisp', '-autoexit']
player = subprocess.Popen(FFPLAY_CMD, stdin=subprocess.PIPE)

class StreamMonitor(FileSystemEventHandler):
    def __init__(self):
        self.last_group_read = {} # Tracks progress for each parquet file
        self.quality_threshold = 0.5

    def on_modified(self, event):
        if not event.is_directory and event.src_path.endswith(".parquet"):
            self.process_new_chunks(event.src_path)

    def process_new_chunks(self, file_path):
        try:
            # 2. PEEK INTO THE GROWING FILE
            pfile = pq.ParquetFile(file_path)
            num_groups = pfile.num_row_groups
            
            start_idx = self.last_group_read.get(file_path, 0)
            
            for i in range(start_idx, num_groups):
                # Only pull what we need: Audio for playback, Score for the Kill-Switch
                table = pfile.read_row_group(i, columns=['audio', 'score'])
                
                audio_bytes = table['audio'][0].as_buffer().to_pybytes()
                score = table['score'][0].as_py()

                # 3. THE JUDGE (Kill Logic)
                if score < self.quality_threshold:
                    print(f"⚠️ Low Quality ({score})! Killing Stream: {file_path}")
                    self.kill_stream()
                    return

                # 4. THE PLAYBACK (Pipe to FFmpeg)
                print(f"▶️ Playing chunk {i} from {os.path.basename(file_path)}...")
                player.stdin.write(audio_bytes)
                player.stdin.flush()

            self.last_group_read[file_path] = num_groups
            
        except Exception as e:
            # Common if file is currently being locked by the writer
            pass

    def kill_stream(self):
        # Stop the player and could also trigger an OS signal to the Agent process
        player.terminate()
        print("🛑 Audio Playback Terminated.")

# 5. INITIALISE THE WATCHDOG
if __name__ == "__main__":
    if not os.path.exists("./streams"): os.makedirs("./streams")
    
    event_handler = StreamMonitor()
    observer = Observer()
    observer.schedule(event_handler, path="./streams", recursive=False)
    
    print("🕵️ Watchdog active. Waiting for Parquet streams...")
    observer.start()
    
    try:
        while True:
            time.sleep(1)
    except KeyboardInterrupt:
        observer.stop()
        player.kill()
    observer.join()
Use code with caution.

Why this setup is elite:
Low Memory Footprint: It uses the PyArrow Columnar Pruning to ignore the text column entirely during playback, focusing only on audio and score.
Decoupled Architecture: If your Agent crashes, the player keeps going until the buffer is empty. If the player crashes, the Agent keeps writing.
OS Agnostic: Using Watchdog's Cross-Platform API ensures this works on Linux, macOS, and Windows.
Instant Quality Control: By checking the score before writing to stdin, the user never hears the "bad" audio.
Pro-Tip: If you find ffplay has too much "jitter," you can add a buffer filter to the command: -af "abitscope,alimiter,aresample=async=1".

def process_audio_pro(audio_bytes):
    """ Feature 4: Smart Trimming & Normalization on i9 CPU """
    # Load from bytes
    y, sr = librosa.load(io.BytesIO(audio_bytes), sr=24000)
    
    # 1. Trim leading/trailing silence (>20db drop)
    y_trimmed, _ = librosa.effects.trim(y, top_db=20)
    
    # 2. Normalize volume to -1.0 to 1.0 range
    y_norm = librosa.util.normalize(y_trimmed)

    fade_out = np.linspace(1, 0, fade_len)

    # Save back to bytes
    buf = io.BytesIO()
    sf.write(buf, y_norm, sr, format='WAV')
    return buf.getvalue()

import json
import requests
import uuid

def queue_beast_canyon_job(text, voice_style, char_img=None, do_lipsync=False):
    # 1. Base Audio Nodes (The 'World's Supply' of Books Path)
    workflow = {
        "3": {
            "inputs": {"text": text, "voice_design": voice_style, "model": "qwen3-tts-1.7b.gguf"},
            "class_type": "Qwen3TTSNode"
        },
        "4": {
            "inputs": {"filename_prefix": f"audio_{uuid.uuid4().hex[:8]}", "audio": ["3", 0]},
            "class_type": "SaveAudio"
        }
    }

    # 2. Optional Video Path (The 'Character Asset' Path)
    if do_lipsync and char_img:
        workflow["5"] = {
            "inputs": {"audio": ["3", 0], "images": ["6", 0], "face_det_batch_size": 4},
            "class_type": "Wav2LipNode"
        }
        workflow["6"] = {
            "inputs": {"image": char_img},
            "class_type": "LoadImage"
        }
        workflow["7"] = {
            "inputs": {"filename_prefix": "video_out", "video": ["5", 0]},
            "class_type": "SaveVideo"
        }

    # 3. Fire to ComfyUI API
    payload = {"prompt": workflow, "client_id": "beast_canyon_agent"}
    response = requests.post(COMFY_URL, data=json.dumps(payload))
    return response.json()
