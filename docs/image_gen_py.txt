from framework.comfyui_generator import ImageGenerator
import os
from agentscope.agent import AgentBase
from agentscope.message import Msg

class ImageGenAgent(AgentBase):
    """
    Agent responsible for generating images via ComfyUI using the new Generator wrapper.
    """
    def __init__(self, name: str, base_url: str, output_dir: str):
        super().__init__()
        self.name = name
        # Parse base_url to host:port for the generator
        host = base_url.replace("http://", "").replace("/", "")
        
        # Load workflow from local file if available, else standard
        workflow_path = os.path.join(os.path.dirname(__file__), "..", "..", "workflows", "generate_Image.json")
        
        self.generator = ImageGenerator(workflow_path=workflow_path, server=host)
        self.output_dir = output_dir

    def reply(self, x: Msg = None) -> Msg:
        # Generate based on content
        prompt = x.content
        
        try:
            # We assume x.content is the positive prompt
            # Negative prompt could be extracted if we had a structured message, but defaulting for now
            negative = "blurry, low quality, watermark, text"
            
            results = self.generator.generate(positive_prompt=prompt, negative_prompt=negative)
            saved_paths = self.generator.save_images(results, self.output_dir)
            
            # Return local path (or URL if we had a media server)
            # For the web UI, we need to serve this. The API currently serves static from frontend/public.
            # We should probably save to frontend/public/assets/generated or similar.
            
            # Hack: return the filename so frontend can construct URL
            filename = os.path.basename(saved_paths[0]) if saved_paths else "error.png"
            
            return Msg(name=self.name, content=f"generated_images/{filename}", role="assistant")
            
        except Exception as e:
            return Msg(name=self.name, content=f"Image generation failed: {e}", role="assistant")
